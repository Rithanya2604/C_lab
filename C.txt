async function searchBook() {
    const book = document.getElementById("bookName").value.trim();
    const results = document.getElementById("results");
    if (!book) {
        results.innerHTML = "<p class='error'>Enter a book name.</p>";
        return;
    }
    results.innerHTML = "<p>Searching books...</p>";
    try {
        const url =
            `https://openlibrary.org/search.json?q=${encodeURIComponent(book)}`;
        const response = await fetch(url);
        if (!response.ok) {
            throw new Error("API request failed");
        }
        const data = await response.json();
        displayBooks(data.docs);
    } catch (error) {
        results.innerHTML =
            "<p class='error'>Unable to retrieve books. Check your internet connection.</p>";
    }
}
function displayBooks(books) {
    const results = document.getElementById("results");
    results.innerHTML = "";
    if (!books || books.length === 0) {
        results.innerHTML = "<p>No books found.</p>";
        return;
    }
    books.slice(0, 10).forEach(book => {
        const title = book.title || "Unknown title";
        const authors = book.author_name
            ? book.author_name.join(", ")
            : "Unknown author";
        const year = book.first_publish_year || "Not available";
        const card = document.createElement("div");
        card.className = "book";
        const heading = document.createElement("h3");
        heading.textContent = title;
        const author = document.createElement("p");
        author.textContent = "Author: " + authors;
        const published = document.createElement("p");
        published.textContent = "Published: " + year;
        card.append(heading, author, published);
        results.appendChild(card);
    });
}

<script type="module">
    import { initializeApp }
        from "https://www.gstatic.com/firebasejs/12.5.0/firebase-app.js";

    import {
        getFirestore, collection, addDoc, getDocs,
        doc, updateDoc, deleteDoc
    } from "https://www.gstatic.com/firebasejs/12.5.0/firebase-firestore.js";

    const firebaseConfig = {
        // enter apikey
    };

    const app = initializeApp(firebaseConfig);
    const db = getFirestore(app);
    const filesRef = collection(db, "files");

    const nameInput = document.getElementById("name");
    const descInput = document.getElementById("description");
    const message = document.getElementById("message");
    const records = document.getElementById("records");
    const saveBtn = document.getElementById("saveBtn");

    let editId = null;

    // CREATE and UPDATE
    saveBtn.addEventListener("click", async () => {
        const name = nameInput.value.trim();
        const description = descInput.value.trim();

        if (!name || !description) {
            message.textContent = "Enter both fields!";
            return;
        }

        try {
            if (editId) {
                await updateDoc(doc(db, "files", editId), {
                    name, description
                });
                message.textContent = "Record updated successfully!";
                editId = null;
                saveBtn.textContent = "Add Record";
            } else {
                await addDoc(filesRef, { name, description });
                message.textContent = "Record added successfully!";
            }

            nameInput.value = "";
            descInput.value = "";
            await loadFiles();
        } catch (error) {
            message.textContent = "Error: " + error.message;
        }
    });

    // READ
    async function loadFiles() {
        records.replaceChildren();

        try {
            const snapshot = await getDocs(filesRef);

            if (snapshot.empty) {
                records.textContent = "No records found.";
                return;
            }

            snapshot.forEach(item => {
                const data = item.data();
                const card = document.createElement("div");
                card.className = "record";

                const title = document.createElement("h3");
                title.textContent = data.name || "";

                const desc = document.createElement("p");
                desc.textContent = data.description || "";

                const editBtn = document.createElement("button");
                editBtn.className = "edit";
                editBtn.textContent = "Edit";

                editBtn.addEventListener("click", () => {
                    nameInput.value = data.name || "";
                    descInput.value = data.description || "";
                    editId = item.id;
                    saveBtn.textContent = "Update Record";
                    message.textContent = "Edit the fields and click Update Record.";
                });

                const deleteBtn = document.createElement("button");
                deleteBtn.className = "delete";
                deleteBtn.textContent = "Delete";

                deleteBtn.addEventListener("click", async () => {
                    if (!confirm("Delete this record?")) return;

                    try {
                        await deleteDoc(doc(db, "files", item.id));
                        message.textContent = "Record deleted successfully!";
                        await loadFiles();
                    } catch (error) {
                        message.textContent = "Delete failed: " + error.message;
                    }
                });

                card.append(title, desc, editBtn, deleteBtn);
                records.appendChild(card);
            });
        } catch (error) {
            message.textContent = "Read failed: " + error.message;
        }
    }

    document.getElementById("viewBtn")
        .addEventListener("click", loadFiles);

    loadFiles();

const express = require("express");
const multer = require("multer");
const { Dropbox } = require("dropbox");
const app = express();
const upload = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 10 * 1024 * 1024 }
});
const dbx = new Dropbox({
  accessToken: process.env.DROPBOX_ACCESS_TOKEN,
  fetch: globalThis.fetch
});
app.use(express.static("public"));
app.use(express.json());
// List files
app.get("/files", async (req, res) => {
  try {
    const result = await dbx.filesListFolder({ path: "" });
    res.json(result.result.entries);
  } catch (err) {
    res.status(500).json({ error: "Unable to list files" });
  }
});
// Upload a file
app.post("/upload", upload.single("file"), async (req, res) => {
  try {
    if (!req.file) {
      return res.status(400).json({ error: "Select a file" });
    }
    await dbx.filesUpload({
      path: "/" + req.file.originalname,
      contents: req.file.buffer,
      mode: { ".tag": "overwrite" }
    });
    res.json({ message: "File uploaded successfully!" });
  } catch (err) {
    res.status(500).json({ error: "Upload failed" });
  }
});
// Delete a file
app.delete("/delete", async (req, res) => {
  try {
    const { path } = req.body;
    if (!path || !path.startsWith("/")) {
      return res.status(400).json({ error: "Invalid file path" });
    }
    await dbx.filesDeleteV2({ path });
    res.json({ message: "File deleted successfully!" });
  } catch (err) {
    res.status(500).json({ error: "Delete failed" });
  }
});
app.listen(3000, () => {
  console.log("Server running at http://localhost:3000");
});

const form = document.getElementById("uploadForm");
const fileInput = document.getElementById("fileInput");
const fileList = document.getElementById("fileList");
const message = document.getElementById("message");

async function loadFiles() {
  fileList.replaceChildren();

  try {
    const response = await fetch("/files");
    if (!response.ok) throw new Error();

    const files = await response.json();

    files.forEach(file => {
      if (file[".tag"] !== "file") return;

      const li = document.createElement("li");
      const name = document.createElement("span");
      name.textContent = `${file.name} (${file.size} bytes)`;
      li.appendChild(name);

      const remove = document.createElement("button");
      remove.textContent = "Delete";
      remove.onclick = async () => {
        if (!confirm("Delete " + file.name + "?")) return;

        const response = await fetch("/delete", {
          method: "DELETE",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ path: file.path_display })
        });

        const data = await response.json();
        message.textContent = data.message || data.error;
        if (response.ok) loadFiles();
      };
      li.appendChild(remove);

      fileList.appendChild(li);
    });

    if (fileList.children.length === 0) {
      const li = document.createElement("li");
      li.textContent = "No files stored yet.";
      fileList.appendChild(li);
    }
  } catch {
    message.textContent = "Could not load files.";
  }
}

form.addEventListener("submit", async event => {
  event.preventDefault();

  const file = fileInput.files[0];
  if (!file) return;

  const data = new FormData();
  data.append("file", file);

  message.textContent = "Uploading...";

  try {
    const response = await fetch("/upload", {
      method: "POST",
      body: data
    });

    const result = await response.json();
    message.textContent = result.message || result.error;

    if (response.ok) {
      form.reset();
      await loadFiles();
    }
  } catch {
    message.textContent = "Upload failed. Check the server.";
  }
});

document.getElementById("refreshBtn")
  .addEventListener("click", loadFiles);

loadFiles();
